def connect_target_excel(file_stream):
    # 经营目标：业务方自助上传，开发只定义"转成什么标准字段"
    df = read_excel(file_stream)

    })
    df["target"] = df["target"] * 10000  # 万元 → 元，口径统一
    return save_to_warehouse(df, "dim_month_target")
3
沉淀语义模型：把业务口径定死 模型层

把销售额、达成率等指标的定义口径写进语义模型，业务方只能选标准指标、不能自己造口径。

from dataclasses import dataclass

@dataclass
class Measure:
    name: str          # gmv / achieve_rate / avg_price
    label: str         # 业务方看到的中文名"销售额"
    formula: str       # 口径公式，写死在这里
    sql: str           # 对应的 SQL 片段

MEASURES = {
    "gmv": Measure("gmv", "销售额 GMV",
        "实付金额 - 退款金额(按退款发生月冲减)",
        "COALESCE(SUM(pay_amt),0) - COALESCE(SUM(refund_amt),0)"),
    "achieve_rate": Measure("achieve_rate", "目标达成率",
        "GMV ÷ 月度目标 × 100%",
        "ROUND(gmv / NULLIF(target,0) * 100, 1)"),
}

def get_measure(name):
    # 指标名必须在 MEASURES 里，否则拒绝——
    # 这就是"只能选标准指标、不能自己造口径"的代码兜底
    if name not in MEASURES:
        raise ValueError("未定义的口径: " + name)
    return MEASURES[name]

注册图表组件：可视化零件库 可视化层

把常用图表封装成标准组件，并定义好下钻、联动等交互，供业务方拖拽使用。

// 开发定义组件契约：数据、配置项、交互事件
interface BarChartProps {
  data: Series[];                 // 语义模型返回的结果集

  measure: string;                // 度量（gmv）
  onDrillDown?: (dimValue) => void;   // 下钻回调
  onFilter?: (filter) => void;        // 联动过滤回调
}

function BarChart(props: BarChartProps) {
  return (
    <div onClick={() => { props.onDrillDown && props.onDrillDown(activeDim); }}>
      {/* 点柱子 → 触发下钻，钻到门店明细 */}

    </div>
  );
}

搭查询引擎与缓存：保证秒开 性能优化

把业务方点选的维度、度量自动拼成 SQL，并加上缓存和超时熔断，保证看板秒级出结果。

def query(dashboard_cfg, cache=True):
    # 1. 根据业务方点选的维度/度量，自动拼出优化过的 SQL
    sql = build_sql(
        dims=dashboard_cfg.dims,          # month, channel
        measures=dashboard_cfg.measures,  # 只会是 MEASURES 里的标准指标
    )

    # 2. 先查缓存：相同看板短时间内重复看，不重复查库
    key = md5(sql + str(dashboard_cfg.filters))
    if cache and redis.exists(key):
        return redis.get(key)

    # 3. 带超时熔断：一个慢查询不能拖垮整个平台
    try:
        result = db.execute(sql, timeout=3.0)
    except QueryTimeout:
        return fallback_cached(key)  # 熔断后读兜底缓存

    redis.set(key, result, ttl=300)
    return result

配置权限与发布：安全可控 服务层

给每个用户配置数据权限，把数据范围自动注入到每条查询，并记录操作日志留痕。

def apply_row_permission(user, sql):
    # 大区经理只看本大区，销售只看自己负责的门店
    scope = get_user_scope(user)   # {region: "华东", stores: [...]}
    if user.role == "region_manager":
        sql += " AND region = '%s'" % scope["region"]
    elif user.role == "sales":
        sql += " AND store_id IN (%s)" % join(scope["stores"])
    return sql

def audit(user, action, target):
    # 谁在什么时候改了什么，全程留痕
    log.insert(user=user.id, action=action,
               target=target, ts=now())

上线运维：持续保障平台稳定 运维

给平台接入监控埋点，实时上报查询性能、可用性和使用情况，慢查询自动告警。

def query_with_metrics(dashboard_cfg):
    t0 = time.time()
    result = query(dashboard_cfg)
    cost = (time.time() - t0) * 1000

    # 三类监控指标自动上报，运维大盘实时可见
    metrics.gauge("dashboard.query_latency_ms", cost)   # 性能
    metrics.incr("dashboard.query_count")              # 使用
    if cost > 1000:
        metrics.incr("dashboard.slow_query_count")       # 慢查询告警
        alert.send("有看板响应 > 1s，请排查")
    return result
